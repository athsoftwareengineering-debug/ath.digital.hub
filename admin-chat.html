📭 No private messages yet.
'; } } catch (error) { console.error('Error loading users:', error); container.innerHTML = '
Error loading users
'; } } function renderUserList(users) { const container = document.getElementById('userList'); if (!container) return; let html = ''; for (const user of users) { const isActive = currentSelectedUser === user.user_id; const unreadBadge = (user.unreadCount > 0) ? `${user.unreadCount}` : ''; html += `
${escapeHtml(user.username)} ${unreadBadge}
🆔 ${escapeHtml(user.user_id)}
`; } container.innerHTML = html; } // When user is selected, mark messages as read async function selectUser(userId, username) { currentSelectedUser = userId; currentSelectedUserName = username; // Mark unread messages as read when selected await markMessagesAsRead(userId); // ... rest of existing code } async function markMessagesAsRead(userId) { try { await fetch(`${API_BASE}/api/chat/private/mark-read/${userId}`, { method: 'PUT', credentials: 'include' }); // Refresh user list to update badge loadUserList(); } catch(e) { console.error('Error marking messages as read:', e); } }
